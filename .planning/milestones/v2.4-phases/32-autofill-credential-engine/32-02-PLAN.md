   const FormDetector = {
     detectedForms: new WeakSet(),
     autofillPaused: false,  // D-13 互斥标志

     scanForLoginForms() {
       // 扫描所有 input[type="password"] 字段
       // 对每个 password 字段，查找关联的 username 字段
       // 两层检测：第一层 password 存在即为登录表单，第二层多字段特征匹配
     },

     findUsernameField(pwField, form) {
       // 在 password 字段前查找 type=text/email 的输入框
       // 优先匹配 name/id/placeholder 包含 user/login/email/account/phone 的字段
       // 兜底：password 字段前最近的 text/email 输入框
     },

     detectFormType(pwField, form) {
       // 检测表单类型：'login' | 'register' | 'password-change'
       // 多个 password 字段 → password-change
       // 包含 register/sign-up/注册 关键词 → register
       // 否则 → login
     },

     extractCredentials(formInfo) {
       // 直接读取 input[type=text/email] 和 input[type=password] 的 value (per D-04)
       return { username: formInfo.usernameField.value, password: formInfo.passwordField.value };
     },

     attachSubmitListener(formInfo) {
       // 监听 form submit 事件 (per D-03)
       // 检查 autofillPaused 标志 (per D-13)
       // 提取凭据并通过 sendToHost 发送
     }
   };

3. 添加自动填充引擎（per 32-RESEARCH.md Pattern 4）：

   const AutofillEngine = {
     fillCredentials(credentials, formInfo) {
       if (FormDetector.autofillPaused) return;  // D-13 互斥
       // 使用 native setter 设置值（per 32-RESEARCH.md Pitfall 2）
       this.setInputValue(formInfo.usernameField, credentials.username);
       this.setInputValue(formInfo.passwordField, credentials.password);
     },

     setInputValue(input, value) {
       // 获取 native setter 避免 React/Vue 拦截
       const nativeSetter = Object.getOwnPropertyDescriptor(
         window.HTMLInputElement.prototype, 'value'
       ).set;
       nativeSetter.call(input, value);
       // 触发 input 和 change 事件
       input.dispatchEvent(new Event('input', { bubbles: true }));
       input.dispatchEvent(new Event('change', { bubbles: true }));
     }
   };

4. 添加 autofill/fillForm 互斥监听（per D-13~D-16, 32-RESEARCH.md Pattern 5）：

   ipcRenderer.on('autofill:pause', () => {
     FormDetector.autofillPaused = true;
   });

   ipcRenderer.on('autofill:resume', () => {
     FormDetector.autofillPaused = false;
   });

5. 页面加载完成和 MutationObserver（per D-01）：

   window.addEventListener('DOMContentLoaded', () => {
     const forms = FormDetector.scanForLoginForms();
     forms.forEach(formInfo => FormDetector.attachSubmitListener(formInfo));
     // 发送 autofill 请求到 renderer
     if (forms.length > 0) {
       ipcRenderer.sendToHost('credential:autofill-request');
     }
   });

   // MutationObserver 持续监听（per D-01）— 需要防抖（per 32-RESEARCH.md Pitfall 3）
   let mutationTimer = null;
   const observer = new MutationObserver(() => {
     if (mutationTimer) clearTimeout(mutationTimer);
     mutationTimer = setTimeout(() => {
       const forms = FormDetector.scanForLoginForms();
       forms.forEach(formInfo => FormDetector.attachSubmitListener(formInfo));
     }, 300);
   });
   observer.observe(document.body, { childList: true, subtree: true });

6. 监听 autofill 填充指令（来自 renderer，由主进程中转）：

   ipcRenderer.on('credential:do-autofill', (event, credentials) => {
     const forms = FormDetector.scanForLoginForms();
     if (forms.length > 0) {
       AutofillEngine.fillCredentials(credentials, forms[0]);
     }
   });
