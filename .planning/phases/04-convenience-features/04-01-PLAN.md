1. **模块结构**：
```javascript
const Store = require('electron-store');
const { URL } = require('url');

const store = new Store({ name: 'assignment-rules' });
```

2. **规则数据模型**：
```javascript
{
  id: string,           // rule-1, rule-2, ...
  containerId: string,  // 目标容器 ID
  pattern: string,      // 匹配模式（域名或通配符）
  enabled: boolean,     // 是否启用
  createdAt: number     // 创建时间戳
}
```

3. **导出函数**：

**getRules():**
- 返回所有规则
- 返回 Array.from(rules.values())

**getRule(ruleId):**
- 返回指定规则
- 返回 rules.get(ruleId) || null

**createRule(containerId, pattern):**
- 验证 pattern 非空
- 验证 containerId 存在
- 生成 ruleId: `rule-${++ruleCounter}`
- 创建规则对象
- 保存到 store
- 返回规则

**updateRule(ruleId, updates):**
- 获取规则
- 合并 updates（pattern, enabled）
- 保存到 store
- 返回更新后的规则

**deleteRule(ruleId):**
- 删除规则
- 保存到 store
- 返回 true

**matchUrl(url):**
- 解析 URL 获取 hostname
- 遍历所有启用的规则
- 检查 hostname 是否匹配 pattern
- 匹配规则：
  - 精确匹配：`google.com` 匹配 `google.com`
  - 通配符匹配：`*.google.com` 匹配 `mail.google.com`
  - 子域名匹配：`google.com` 匹配 `www.google.com`
- 返回匹配的容器 ID 或 null

4. **URL 匹配逻辑**：
```javascript
function matchUrl(url) {
  try {
    const parsed = new URL(url);
    const hostname = parsed.hostname;
    
    const rules = getRules().filter(r => r.enabled);
    
    for (const rule of rules) {
      if (matchPattern(hostname, rule.pattern)) {
        return rule.containerId;
      }
    }
    
    return null;
  } catch (error) {
    return null;
  }
}

function matchPattern(hostname, pattern) {
  // 精确匹配
  if (hostname === pattern) {
    return true;
  }
  
  // 通配符匹配
  if (pattern.startsWith('*.')) {
    const domain = pattern.slice(2);
    return hostname.endsWith('.' + domain) || hostname === domain;
  }
  
  // 子域名匹配
  return hostname.endsWith('.' + pattern);
}
```

- 合并默认配置和用户自定义配置

- 返回指定操作的快捷键

**setShortcut(action, accelerator):**
- 更新快捷键配置
- 保存到 store
- 重新注册快捷键

**registerShortcuts(window):**
- 注册所有全局快捷键
- 每个快捷键触发时发送 IPC 消息到渲染进程
- 日志：`[Realm] 注册快捷键: ${action} -> ${accelerator}`

**unregisterAll():**
- 注销所有全局快捷键
- 日志：`[Realm] 注销所有快捷键`

4. **快捷键处理**：
```javascript
function registerShortcuts(window) {
  const shortcuts = getShortcuts();
  
  Object.entries(shortcuts).forEach(([action, accelerator]) => {
    if (globalShortcut.isRegistered(accelerator)) {
      console.log(`[Realm] 快捷键已注册，跳过: ${accelerator}`);
      return;
    }
    
    const ret = globalShortcut.register(accelerator, () => {
      console.log(`[Realm] 快捷键触发: ${action} (${accelerator})`);
      
      // 发送 IPC 消息到渲染进程
      if (window && !window.isDestroyed()) {
        window.webContents.send('shortcut:triggered', action);
      }
    });
    
    if (!ret) {
      console.error(`[Realm] 快捷键注册失败: ${accelerator}`);
    }
  });
}
```
