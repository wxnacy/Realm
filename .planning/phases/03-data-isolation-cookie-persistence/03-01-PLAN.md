```javascript
async function saveCookies(containerId) {
  const ses = session.fromPartition(`persist:container-${containerId}`);
  const cookies = await ses.cookies.get({});

  const formattedCookies = cookies.map(cookie => ({
    name: cookie.name,
    value: cookie.value,
    domain: cookie.domain,
    path: cookie.path,
    expirationDate: cookie.expirationDate,
    secure: cookie.secure,
    httpOnly: cookie.httpOnly,
    sameSite: cookie.sameSite,      // D-05
    hostOnly: cookie.hostOnly,      // D-06
    url: cookie.url                 // 用于 set 操作
  }));

  ensureCookieDir();
  const filePath = path.join(COOKIE_DIR, `${containerId}.json`);
  fs.writeFileSync(filePath, JSON.stringify(formattedCookies, null, 2));

  console.log(`[Realm] 保存容器 Cookie: ${containerId} (${cookies.length} 个)`);
  return { success: true, count: cookies.length };
}
```

3. **修改 loadCookies 函数**：
```javascript
async function loadCookies(containerId) {
  const filePath = path.join(COOKIE_DIR, `${containerId}.json`);
  if (!fs.existsSync(filePath)) {
    return { success: true, count: 0 };
  }

  const cookies = JSON.parse(fs.readFileSync(filePath, 'utf8'));
  const ses = session.fromPartition(`persist:container-${containerId}`);

  for (const cookie of cookies) {
    try {
      await ses.cookies.set({
        url: cookie.url || `http${cookie.secure ? 's' : ''}://${cookie.domain.replace(/^\./, '')}${cookie.path}`,
        name: cookie.name,
        value: cookie.value,
        domain: cookie.domain,
        path: cookie.path,
        expirationDate: cookie.expirationDate,
        secure: cookie.secure,
        httpOnly: cookie.httpOnly,
        sameSite: cookie.sameSite,
        hostOnly: cookie.hostOnly
      });
    } catch (err) {
      console.error(`[Realm] 加载 Cookie 失败: ${cookie.name}@${cookie.domain}`, err.message);
    }
  }

  console.log(`[Realm] 加载容器 Cookie: ${containerId} (${cookies.length} 个)`);
  return { success: true, count: cookies.length };
}
```

```javascript
/**
 * 删除容器的 Cookie 文件和 Session 数据
 * @param {string} containerId - 容器 ID
 * @returns {Object} 操作结果
 */
function deleteCookies(containerId) {
  const filePath = path.join(COOKIE_DIR, `${containerId}.json`);

  // D-01: 删除 Cookie 文件
  if (fs.existsSync(filePath)) {
    fs.unlinkSync(filePath);
    console.log(`[Realm] 删除容器 Cookie 文件: ${containerId}`);
  }

  // D-02: 清理 Session 数据
  const ses = session.fromPartition(`persist:container-${containerId}`);
  ses.clearStorageData();

  return { success: true };
}
```
